In this stage, you'll implement the /echo/{str} endpoint, which accepts a string and returns it in the response body.

Response body
A response body is used to return content to the client. This content may be an entire web page, a file, a string, or anything else that can be represented with bytes.

Your /echo/{str} endpoint must return a 200 response, with the response body set to given string, and with a Content-Type and Content-Length header.

Here's an example of an /echo/{str} request:

GET /echo/abc HTTP/1.1\r\nHost: localhost:4221\r\nUser-Agent: curl/7.64.1\r\nAccept: */*\r\n\r\n
And here's the expected response:

HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: 3\r\n\r\nabc
Here's a breakdown of the response:

// Status line
HTTP/1.1 200 OK
\r\n                          // CRLF that marks the end of the status line

// Headers
Content-Type: text/plain\r\n  // Header that specifies the format of the response body
Content-Length: 3\r\n         // Header that specifies the size of the response body, in bytes
\r\n                          // CRLF that marks the end of the headers

// Response body
abc                           // The string from the request
The two headers are required for the client to be able to parse the response body. Note that each header ends in a CRLF, and the entire header section also ends in a CRLF.

Tests
The tester will execute your program like this:

$ ./your_program.sh
The tester will then send a GET request to the /echo/{str} endpoint on your server, with some random string.

$ curl -v http://localhost:4221/echo/abc
Your server must respond with a 200 response that contains the following parts:

Content-Type header set to text/plain.
Content-Length header set to the length of the given string.
Response body set to the given string.
HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: 3\r\n\r\nabc
Notes
For more information about HTTP responses, see the MDN Web Docs on HTTP responses or the HTTP/1.1 specification.

int main(int argc, char *argv[]) {
  // Check if any option was passed (directory), and if it was valid
  bool directory;
  if (argc >= 3 && strcmp(argv[1], "--directory") == 0) {
    directory = true;

    // Chech if the directory is valid
    size_t dir_len = strlen(argv[2]);
    if ((dir_len > 0 && argv[2][dir_len - 1] == '/')) {
      printf("Directory option good.\n");
    } else {
      fprintf(stderr,
              "\033[31mInvalid directory. Ingoring the argument.\033[0m\n");
      directory = false;
    }
  } else {
    directory = false;
  }

  // Disable output buffering
  setbuf(stdout, NULL);
  setbuf(stderr, NULL);

  int server_fd;
  server_fd = socket(AF_INET, SOCK_STREAM, 0);
  if (server_fd == -1) {
    fprintf(stderr, "\033[31mSocket creation failed: %s\033[0m\n",
            strerror(errno));
    return 1;
  }

  // Ensures no 'Address already in use' errors when restarting program multiple
  // times
  int reuse = 1;
  if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &reuse, sizeof(reuse)) ==
      -1) {
    fprintf(stderr, "\033[31mSO_REUSEPORT failed: %s\033[0m\n",
            strerror(errno));
    return 1;
  }

  struct sockaddr_in serv_addr = {
      .sin_family = AF_INET,
      .sin_port = htons(PORT),
      .sin_addr = {htonl(INADDR_ANY)},
  };

  if (bind(server_fd, (struct sockaddr *)&serv_addr, sizeof(serv_addr)) != 0) {
    fprintf(stderr, "\033[31mBind failed: %s\033[0m\n", strerror(errno));
    return 1;
  }

  int connection_backlog = 10;
  if (listen(server_fd, connection_backlog) != 0) {
    fprintf(stderr, "\033[31mLalalalala Listen failed: %s\033[0m\n",
            strerror(errno));
    return 1;
  }

  while (1) {
    struct sockaddr_in client_addr;
    socklen_t client_addr_len = sizeof(client_addr);

    printf("Waiting for a client to connect...\n");

    int client_fd =
        accept(server_fd, (struct sockaddr *)&client_addr, &client_addr_len);
    if (client_fd == -1) {
      fprintf(stderr, "\033[31mFailed to connect: %s\033[0m\n",
              strerror(errno));
      return 1; // TODO handle failed connections in a way that doesn't crash
                // the server in case of an error
    }

    printf("Client connected\n");

    char request[BUFFER_SIZE];
    char res[BUFFER_SIZE];

    if (!fork()) {
      // Fill buffer with received message
      ssize_t bytes_received = recv(client_fd, request, BUFFER_SIZE - 1, 0);
      if (bytes_received == -1) {
        fprintf(stderr, "\033[31mReceive failed: %s\033[0m\n", strerror(errno));
        close(client_fd);
        exit(1);
      }

      // Process request and send response
      close(server_fd);
      request[bytes_received] = '\0';
      process_request(argc, argv, client_fd, request, res, directory);

      close(client_fd);
      exit(0);
    }
    close(client_fd);
  }

  close(server_fd);
  return 0;
}

void process_request(int argc, char *argv[], int client_fd, char *request,
                     char *res, bool directory) {
  // To encode or not to encode
  bool encode = false;
  char *ae = extract_header_value(request, "Accept-Encoding:");
  char *needle;
  if (ae != NULL && (needle = strstr(ae, "gzip")) != NULL) {
    // Check if the encoding option is really gzip and not gzipsomething
    if (needle[4] == '\0' || needle[4] == ',') {
      encode = true;
    }
  }

  char *final_content;

  // Extract path from GET request
  char *path;
  char *request_tail; // Points to the rest of the request char array after each
                      // tokenize
  char *method;
  method = strtok_r(request, " ", &request_tail);
  path = strtok_r(NULL, " ", &request_tail);

  // Respond based on path
  if (strcmp(path, "/") == 0) {
    // build_response(res, BUFFER_SIZE, HTTP_200_OK, "\r\n", "", false);
    send_response(client_fd, HTTP_200_OK, 0, NULL, 1, "", false);

  } else if (strncmp(path, "/echo/", 6) == 0) {
    // Extract content
    char *content = strtok(path, "/");
    content = strtok(NULL, "/");

    // In case the path was /echo/, which leads to segfault
    if (content == NULL) {
      // Treat it as an echo with empty content
      char *headers[] = {CONTENT_TYPE_TEXT};
      send_response(client_fd, HTTP_200_OK, 1, headers, 0, "", false);
    } else {
      char *headers[] = {CONTENT_TYPE_TEXT};
      send_response(client_fd, HTTP_200_OK, 1, headers, strlen(content),
                    content, encode);
    }

  } else if (strcmp(path, "/user-agent") == 0) {
    char *user_agent = extract_header_value(request_tail, "User-Agent:");
    char *headers[] = {CONTENT_TYPE_TEXT};
    send_response(client_fd, HTTP_200_OK, 1, headers, strlen(user_agent),
                  user_agent, encode);

  } else if (strncmp(path, "/files/", 7) == 0 && strcmp(method, "GET") == 0 &&
             directory) {
    get_file(client_fd, path, argv[2], res, encode);

  } else if (strncmp(path, "/files/", 7) == 0 && strcmp(method, "POST") == 0 &&
             directory) {
    create_file(client_fd, path, argv[2], res, request_tail);

  } else {
    send_response(client_fd, HTTP_404_NOT_FOUND, 0, NULL, 1, "", false);
  }
}

void create_file(int client_fd, char *path, char *directory, char *res,
                 char *request) {
  // Extract filename
  char *file = strtok(path, "/");
  file = strtok(NULL, "/");

  // Full path
  char directory_full[BUFFER_SIZE];
  size_t dir_len = strlen(directory);
  snprintf(directory_full, sizeof(directory_full), "%s%s", directory, file);

  // Extract content
  char *body = strstr(request, "\r\n\r\n");
  if (body != NULL) {
    body += 4; // Move past the "\r\n\r\n" sequence
  } else {
    body = "";
  }

  FILE *fh;
  fh = fopen(directory_full, "a");
  if (fh == NULL) {
    send_response(client_fd, HTTP_404_NOT_FOUND, 0, NULL, 0, "", false);
    fprintf(stderr, "\033[31mInvalid directory.\033[0m\n");
    exit(1);
  }
  fprintf(fh, "%s", body);
  fclose(fh);

  send_response(client_fd, HTTP_201_CREATED, 0, NULL, 0, "", false);
}

void get_file(int client_fd, char *path, char *directory, char *res,
              bool encode) {
  // Extract filename
  char *file = strtok(path, "/");
  file = strtok(NULL, "/");

  // Full path
  char directory_full[BUFFER_SIZE];
  snprintf(directory_full, BUFFER_SIZE, "%s%s", directory, file);

  // Open file
  FILE *fh;
  fh = fopen(directory_full, "r");
  if (fh != NULL) {
    // Read the file into a temporary buffer.
    char file_buffer[BUFFER_SIZE];
    size_t bytes_read = fread(file_buffer, 1, BUFFER_SIZE - 1, fh);
    fclose(fh);
    file_buffer[bytes_read] = '\0';

    // Create the HTTP response.
    char *final_content =
        (char *)malloc(bytes_read); // TODO, check if it's safe to do this
    char *headers[] = {CONTENT_TYPE_APP};
    send_response(client_fd, HTTP_200_OK, 1, headers, bytes_read, file_buffer,
                  encode);

  } else {
    send_response(client_fd, HTTP_200_OK, 0, NULL, 0, "", encode);
  }
}

char *extract_header_value(char *request, char *header) {
  // Duplicate the request to avoid destroying the original
  char *request_dup = strdup(request);
  if (request_dup == NULL) {
    return NULL;
  }

  char *saveptr;
  char *line = strtok_r(request_dup, "\r\n", &saveptr);
  char *header_value = NULL;

  while (line != NULL) {
    if (strncmp(line, header, strlen(header)) == 0) {

      // Found the header; skip header name and whitespace
      header_value = line + strlen(header);
      while (*header_value == ' ' || *header_value == '\t') {
        header_value++;
      }

      break;
    }
    line = strtok_r(NULL, "\r\n", &saveptr);
  }
  // Allocate a copy of the result if found.
  char *header_return_value = header_value ? strdup(header_value) : NULL;
  free(request_dup);
  return header_return_value;
}

int gzip_compress(char *dest, size_t dest_len, char *source) {

  int ret;
  z_stream strm;

  // Initialise z_stream struct
  strm.zalloc = Z_NULL;
  strm.zfree = Z_NULL;
  strm.opaque = Z_NULL;
  strm.avail_in = (uInt)strlen(source);
  strm.next_in = (Bytef *)source;
  strm.avail_out = (uInt)dest_len;
  strm.next_out = (Bytef *)dest;

  // Initialise deflate for gzip compression (15 | 16 produces gzip
  // header/trailer)
  ret = deflateInit2(&strm, Z_DEFAULT_COMPRESSION, Z_DEFLATED, 15 | 16, 8,
                     Z_DEFAULT_STRATEGY);
  if (ret != Z_OK) {
    fprintf(stderr, "\033[31mDeflate initialize error: %d\033[0m\n", ret);
    return -1;
  }

  // Deflate, assumes the entire source fits in one pass
  // TODO use loop to avoid assumption
  ret = deflate(&strm, Z_FINISH);
  if (ret != Z_STREAM_END) {
    fprintf(stderr, "\033[31mDeflate error: %d\033[0m\n", ret);
    deflateEnd(&strm); // Clean up before returning
    return -1;
  }

  // Clean up and return
  (void)deflateEnd(&strm);
  return strm.total_out;
}

void send_response(int client_fd, char *response_code, size_t headers_num,
                   char *headers[], size_t content_size, char *content,
                   bool encode) {
  // Send response code
  send_buffer(client_fd, response_code, strlen(response_code));

  char content_encoded[BUFFER_SIZE];
  if (encode) {
    // Encode the content (gzip compression)
    content_size = gzip_compress(content_encoded, BUFFER_SIZE, content);
    if (content_size == -1) {
      fprintf(stderr, "\033[31mCompression failed: %s\033[0m\n",
              strerror(errno));
      exit(1);
    }
  }

  // Send headers, but first check if there are any
  if (headers_num > 0 && headers != NULL) {
    int i;
    for (i = 0; i < headers_num; i++) {
      send_buffer(client_fd, headers[i], strlen(headers[i]));
    }

    // Send content length header
    char content_length[BUFFER_SIZE];
    snprintf(content_length, BUFFER_SIZE, CONTENT_LENGTH, content_size);
    send_buffer(client_fd, content_length, strlen(content_length));
  }

  // Send CRLF (end of headers)
  send_buffer(client_fd, "\r\n", 2);

  if (encode) {
    // Send encoded content
    send_buffer(client_fd, content_encoded, content_size);

  } else {
    // Send content
    send_buffer(client_fd, content, content_size);
  }
}

void send_buffer(int client_fd, char *buffer, size_t buffer_size) {
  // Send a buffer with error handling
  if (send(client_fd, buffer, buffer_size, 0) == -1) {
    fprintf(stderr, "\033[31mSend failed: %s\033[0m\n", strerror(errno));
    close(client_fd);
    exit(1);
  }
}
