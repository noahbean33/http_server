In this stage, you'll implement the /user-agent endpoint, which reads the User-Agent request header and returns it in the response body.

The User-Agent header
The User-Agent header describes the client's user agent.

Your /user-agent endpoint must read the User-Agent header, and return it in your response body. Here's an example of a /user-agent request:

// Request line
GET
/user-agent
HTTP/1.1
\r\n

// Headers
Host: localhost:4221\r\n
User-Agent: foobar/1.2.3\r\n  // Read this value
Accept: */*\r\n
\r\n

// Request body (empty)
Here is the expected response:

// Status line
HTTP/1.1 200 OK               // Status code must be 200
\r\n

// Headers
Content-Type: text/plain\r\n
Content-Length: 12\r\n
\r\n

// Response body
foobar/1.2.3                  // The value of `User-Agent`
Tests
The tester will execute your program like this:

$ ./your_program.sh
The tester will then send a GET request to the /user-agent endpoint on your server. The request will have a User-Agent header.

$ curl -v --header "User-Agent: foobar/1.2.3" http://localhost:4221/user-agent
Your server must respond with a 200 response that contains the following parts:

Content-Type header set to text/plain.
Content-Length header set to the length of the User-Agent value.
Message body set to the User-Agent value.
HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: 12\r\n\r\nfoobar/1.2.3
Notes
Header names are case-insensitive.

#define RES_SIZE 2048

char *defined_headers[] = {"Content-Type", "Host", "User-Agent", "Accept", "Content-Length"};
typedef struct response
{
	char *http_ver;
	int code;
	char *message;
	char **headers;
	char *body;
} Response;

Response *new_response()
{
	Response *res = (Response *)malloc(sizeof(Response));
	res->http_ver = "HTTP/1.1";
	res->headers = (char **)calloc(HEADER_LEN, sizeof(char *));
	return res;
}

char *toString(Response *res)
{
	char *resp_str = (char *)calloc(RES_SIZE, sizeof(char));
	sprintf(resp_str, "%s %d %s\r\n", res->http_ver, res->code, res->message);
	char header[40];
	for (int i = 0; i < HEADER_LEN; i++)
	{
		if (res->headers[i])
		{
			sprintf(header, "%s: %s\r\n", defined_headers[i], res->headers[i]);
			strcat(resp_str, header);
		}
	}
	strcat(resp_str, "\r\n");
	if (res->body)
		strcat(resp_str, res->body);
	return resp_str;
}

void write_to_res(Response *res, char *content)
{
	res->headers[get_header_idx("Content-Type")] = "text/plain";
	char *len = (char *)calloc(10, sizeof(char));
	sprintf(len, "%d", (int)strlen(content));
	res->headers[get_header_idx("Content-Length")] = len;
	res->body = content;
}

void router(Request *req, int conn)
{
	int m = strlen(req->method);
	int n = strlen(req->url);
	char *action = (char *)calloc(m + n + 2, sizeof(char));
	strcpy(action, req->method);
	action[m] = ' ';
	action[m + 1] = '\0';
	strcat(action, req->url);
	printf("action %s\n", action);
	Response *res = new_response();
	if (strcmp(action, "GET /") == 0)
	{
		res->code = 200;
		res->message = "OK";
		char *resp_str = toString(res);
		write(conn, resp_str, strlen(resp_str));
	}
	else if (strncmp(action, "GET /echo/", 10) == 0)
	{
		char params_id[40];
		strcpy(params_id, action + 10);
		res->code = 200;
		res->message = "OK";
		write_to_res(res, params_id);
		char *resp_str = toString(res);
		write(conn, resp_str, strlen(resp_str));
	}
	else if (strcmp(action, "GET /user-agent") == 0)
	{
		res->code = 200;
		res->message = "OK";
		write_to_res(res, req->headers[get_header_idx("User-Agent")]);
		char *resp_str = toString(res);
		write(conn, resp_str, strlen(resp_str));
	}
	else
	{
		res->code = 404;
		res->message = "Not Found";
		char *resp_str = toString(res);
		write(conn, resp_str, strlen(resp_str));
	}
	free(res);
}

		return 1;

	printf("Waiting for a client to connect...\n");
	client_addr_len = sizeof(client_addr);

	int client_fd = accept(server_fd, (struct sockaddr *)&client_addr, &client_addr_len);
	printf("Client connected\n");
	char *ok_msg = "HTTP/1.1 200 OK\r\n\r\n";
	char req_str[2048];
	read(client_fd, req_str, 2048);
	Request *req = parse_request(req_str);
