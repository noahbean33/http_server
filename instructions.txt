In this stage, you'll add support for concurrent connections.

Tests
The tester will execute your program like this:

$ ./your_program.sh
Then, the tester will create multiple concurrent TCP connections to your server. (The exact number of connections is determined at random.) After that, the tester will send a single GET request through each of the connections.

$ (sleep 3 && printf "GET / HTTP/1.1\r\n\r\n") | nc localhost 4221 &
$ (sleep 3 && printf "GET / HTTP/1.1\r\n\r\n") | nc localhost 4221 &
$ (sleep 3 && printf "GET / HTTP/1.1\r\n\r\n") | nc localhost 4221 &
Your server must respond to each request with the following response:

HTTP/1.1 200 OK\r\n\r\n

int setup_server(int *server_fd, int *client_fd, int *client_addr_len,
                 struct sockaddr_in *client_addr) {

  //
  *server_fd = socket(AF_INET, SOCK_STREAM, 0);
  if (*server_fd == -1) {
    printf("Socket creation failed: %s...\n", strerror(errno));
    exit(1);
  }
  //
  // // Since the tester restarts your program quite often, setting SO_REUSEADDR
  // // ensures that we don't run into 'Address already in use' errors
  int reuse = 1;
  if (setsockopt(*server_fd, SOL_SOCKET, SO_REUSEADDR, &reuse, sizeof(reuse)) <
      0) {
    printf("SO_REUSEADDR failed: %s \n", strerror(errno));
    return 1;
  }
  //
  struct sockaddr_in serv_addr = {
      .sin_family = AF_INET,
      .sin_port = htons(4221),
      .sin_addr = {htonl(INADDR_ANY)},
  };
  //
  if (bind(*server_fd, (struct sockaddr *)&serv_addr, sizeof(serv_addr)) != 0) {
    printf("Bind failed: %s \n", strerror(errno));
    return 1;
  }
  //
  int connection_backlog = 5;
  if (listen(*server_fd, connection_backlog) != 0) {
    printf("Listen failed: %s \n", strerror(errno));
    return 1;
  }
  //
  printf("Waiting for a client to connect...\n");
  *client_addr_len = sizeof(*client_addr);

  return 0;
}

int append_crlf(const dynarr *arr) {
  char *crlf = "\r\n";
  return dyn_push(arr, crlf, strlen(crlf));
}

bool has_crlfcrlf(const char *buf, ssize_t len) {
  if (len < 4) {
    return false;
  }

  for (size_t i = 0; i + 3 < len; i++) {
    if (buf[i] == '\r' && buf[i + 1] == '\n' && buf[i + 2] == '\r' &&
        buf[i + 3] == '\n') {
      return true;
    }
  }

  return false;
}

ssize_t find_header_end(const char *buf, size_t len) {

  for (size_t i = 0; i + 3 < len; i++) {
    if (buf[i] == '\r' && buf[i + 1] == '\n' && buf[i + 2] == '\r' &&
        buf[i + 3] == '\n') {
      return (ssize_t)(i + 4);
    }
  }

  return -1;
}

size_t parse_content_length(const char *buf, size_t header_end) {
  const char *p = buf;
  const char *end = buf + header_end;

  while (p < end) {
    if (strncasecmp(p, "Content-Length:", 15) == 0) {
      p += 15;
      while (*p == ' ' || *p == '\t')
        p++;
      return (size_t)strtoul(p, NULL, 10);
    }

    const char *nl = memchr(p, '\n', end - p);
    if (!nl)
      break;
    p = nl + 1;
  }

  return 0;
}

int read_req(dynarr *req, int server_fd) {

  char buf[BLOCK];

  while (1) {
    ssize_t bytes_read = recv(server_fd, buf, sizeof(buf), 0);

    if (bytes_read == 0) {
      break;
    }

    if (bytes_read == -1) {
      break;
    }

    dyn_push(req, buf, bytes_read);
    // printf("req is %s\n", req->ptr);

    if (has_crlfcrlf(req->ptr, req->len)) {
      break;
    }
  }

  // printf("did we break");

  ssize_t header_end = find_header_end(req->ptr, req->len);
  if (header_end < 0)
    return -1;

  size_t body_len = parse_content_length(req->ptr, header_end);

  while (req->len < (size_t)header_end + body_len) {
    ssize_t n = recv(server_fd, buf, sizeof(buf), 0);

    if (n <= 0)
      return -1;

    if (dyn_push(req, buf, n) < 0)
      return -1;
  }

  return 1;
}

void *handle_client(void *arg) {
  int client_fd = *(int *)arg;
  free(arg); // allocated in main thread

  dynarr req, resp;
  dynarr_init(&req, 1);
  dynarr_init(&resp, 1);

  if (read_req(&req, client_fd) > 0) {
    parse_http_req(&resp, &req);
    send_response(client_fd, &resp);
  }

  free_dynarr(&req);
  free_dynarr(&resp);
  close(client_fd);

  return NULL;
}

int main() {
  // Disable output buffering
  setbuf(stdout, NULL);
  setbuf(stderr, NULL);

  // You can use print statements as follows for debugging, they'll be visible
  // when running tests.
  printf("Logs from your program will appear here!\n");

  int server_fd, client_fd, client_addr_len;
  struct sockaddr_in client_addr;

  if (setup_server(&server_fd, &client_fd, &client_addr_len, &client_addr) ==
      1) {
    close(server_fd);
    return 1;
  }

  while (1) {
    // socklen_t client_addr_len = sizeof(client_addr);

    int client_fd =
        accept(server_fd, (struct sockaddr *)&client_addr, &client_addr_len);

    if (client_fd < 0)
      continue;

    pthread_t tid;

    int *pclient = malloc(sizeof(int));
    *pclient = client_fd;

    pthread_create(&tid, NULL, handle_client, pclient);
    pthread_detach(tid); // no need to join
  }
  return 0;
}
